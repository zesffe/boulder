Bolder is yours to discover.

Best to generate a better, bolder font-family by the traditional method.

require $dir . '/class/bolder.php';

$fam = new bolder();

$fam->add([
	'family' => 'IBMPlexSans',
	'path' => '/fonts',
	'format' => 'opentype',
	'styles' => ['normal', 'italic', ],
	'weights' => ['medium' => 500, 'bold' => 700, ],
]);

echo $fam->export();
